 webpush "github.com/SherClockHolmes/webpush-go"

)

func TestAuditPushAcceptsLoopback(t *testing.T) {
 hit := false
 target := httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
  hit = true
  if r.Method != "POST" { t.Errorf("method: %s", r.Method) }
  w.WriteHeader(http.StatusCreated)
 }))
 defer target.Close()
 key, err := ecdh.P256().GenerateKey(rand.Reader)
 if err != nil { t.Fatal(err) }
 private, public, err := webpush.GenerateVAPIDKeys()
 if err != nil { t.Fatal(err) }
 resp, err := webpush.SendNotification([]byte("audit-only"), &webpush.Subscription{
  Endpoint: target.URL,
  Keys: webpush.Keys{P256dh: base64.RawURLEncoding.EncodeToString(key.PublicKey().Bytes()), Auth: base64.RawURLEncoding.EncodeToString(make([]byte,16))},
 }, &webpush.Options{Subscriber:"mailto:audit@example.com", VAPIDPublicKey:public, VAPIDPrivateKey:private, TTL:60, Urgency:webpush.UrgencyHigh})
 if err != nil { t.Fatal(err) }
 resp.Body.Close()
 if !hit { t.Fatal("loopback request not received") }
 t.Log("Unrestricted Web Push options used by the application successfully sent POST to loopback")
}

func TestAuditLogoutTokenStillValid(t *testing.T) {
 secret := []byte("audit-only-012345678901234567890123456789")
 s := &Server{secret:secret}
 token, err := authjwt.SignAccess(secret, uuid.New(), 24*time.Hour)
 if err != nil { t.Fatal(err) }
 req := httptest.NewRequest(http.MethodPost,"/api/v1/auth/logout",nil)
 req.AddCookie(&http.Cookie{Name:authAccessCookieName,Value:token})
 s.handleLogout(httptest.NewRecorder(),req)
 if _,_,ok := s.principalForAccess(t.Context(),token); !ok { t.Fatal("token rejected") }
 t.Log("The same access token remains accepted after logout")
}
